Python example · launcher.yaml
launcher.yaml 
schemaVersion: "0.2"
id: example.python-webapp
version: "1.0.0"
name: Python Web App
environment:
  mode: existing
parameters:
  port:
    default: 3000
install:
  - program: uv
    args: [sync, --frozen, --no-dev]
service:
  protocol: http-api
  run:
    program: uv
    args:
      - run
      - --frozen
      - --no-sync
      - python
      - server.py
      - --host
      - "${host}"
      - --port
      - "${port}"
  healthPath: /healthz
